const GEOM_TYPES = ['Point', 'MultiPoint', 'LineString', 'MultiLineString', 'Polygon', 'MultiPolygon', 'GeometryCollection'];
function geojsonOf(e) {
  return memo(e, 'geojson', () => {
    let o;
    try { o = JSON.parse(strictText(e)); } catch (err) { throw new Error(/UTF-8/.test(reasonOf(err)) ? reasonOf(err) : `The file isn’t valid JSON: ${reasonOf(err)}`); }
    if (!o || typeof o !== 'object' || Array.isArray(o)) throw new Error('The file’s top level isn’t an object with a "type", so it holds no features.');
    let feats;
    if (o.type === 'FeatureCollection') {
      if (!Array.isArray(o.features)) throw new Error('The file says it is a FeatureCollection but has no "features" list.');
      feats = o.features;
    } else if (o.type === 'Feature') feats = [o];
    else if (GEOM_TYPES.includes(o.type)) feats = [{ type: 'Feature', geometry: o, properties: null }];
    else throw new Error(`The file’s top-level "type" is ${JSON.stringify(o.type === undefined ? null : o.type)}, not FeatureCollection, Feature or a geometry, so it holds no features.`);
    const features = feats.map((f, i) => {
      const where = `Feature ${i + 1}`;
      if (!f || typeof f !== 'object' || f.type !== 'Feature') throw new Error(`${where} in the "features" list doesn’t have "type": "Feature".`);
      if (!('geometry' in f)) throw new Error(`${where} has no "geometry" member.`);
      if (f.geometry !== null) checkGeom(f.geometry, where, 0);
      if (f.properties !== undefined && f.properties !== null && (typeof f.properties !== 'object' || Array.isArray(f.properties))) throw new Error(`${where} has "properties" that aren’t an object.`);
      return { props: f.properties || null, id: f.id, geom: f.geometry };
    });
    let code = 4326; // the coordinate system these files use unless they name another
    const nm = o.crs && o.crs.properties && typeof o.crs.properties.name === 'string' ? o.crs.properties.name : '';
    if (nm) { const m = /EPSG:{1,2}(\d+)\s*$/i.exec(nm); code = /CRS84$/i.test(nm) ? 4326 : m ? +m[1] : null; }
    return { features, crs: crsOf(code, '', undefined, nm) };
  });
}
function checkGeom(g, where, depth) {
  if (!g || typeof g !== 'object' || !GEOM_TYPES.includes(g.type)) throw new Error(`${where} has a geometry whose "type" isn’t one of ${GEOM_TYPES.join(', ')}.`);
  if (g.type === 'GeometryCollection') {
    if (!Array.isArray(g.geometries)) throw new Error(`${where} has a GeometryCollection with no "geometries" list.`);
    if (depth > 16) throw new Error(`${where} nests geometry collections too deeply.`);
    g.geometries.forEach((x) => checkGeom(x, where, depth + 1));
    return;
  }
  const levels = { Point: 0, MultiPoint: 1, LineString: 1, MultiLineString: 2, Polygon: 2, MultiPolygon: 3 }[g.type];
  const pos = (p) => Array.isArray(p) && p.length >= 2 && p.every((v) => typeof v === 'number' && isFinite(v));
  const walk = (c, l) => {
    if (l === 0) { if (!pos(c)) throw new Error(`${where} has a ${g.type} with a position that isn’t a list of two or more numbers.`); return; }
    if (!Array.isArray(c)) throw new Error(`${where} has a ${g.type} whose "coordinates" aren’t nested as that kind of geometry needs.`);
    c.forEach((x) => walk(x, l - 1));
  };
  if (g.type === 'Point' && Array.isArray(g.coordinates) && g.coordinates.length === 0) return; // an empty point
  walk(g.coordinates, levels);
}

const MERC = new Set([3857, 900913, 3785, 102100, 102113]);
const GEOG = new Set([4326, 4269, 4258, 4283, 4617, 4167, 4230, 4267, 4674, 4612, 4979, 4937, 4152, 4759]);
function unitOfWkt(def) { // the length unit a projected system's description names (not the one inside its base system)
  if (!def) return null;
  let s = String(def);
  for (const key of ['GEOGCS[', 'BASEGEOGCRS[', 'BASEGEODCRS[', 'GEOGCRS[']) {
    const i = s.toUpperCase().indexOf(key);
    if (i <= 0) continue;
    let d = 0, j = i + key.length - 1;
    for (; j < s.length; j++) { if (s[j] === '[') d++; else if (s[j] === ']' && --d === 0) break; }
    s = s.slice(0, i) + s.slice(j + 1);
  }
  const m = /(?:LENGTHUNIT|UNIT)\s*\[\s*"([^"]+)"\s*,\s*([0-9.eE+-]+)/i.exec(s);
  return m && +m[2] > 0 ? [m[1], +m[2]] : null;
}
function unitOfCode(c) {
  if (c === null) return null;
  if ((c >= 32601 && c <= 32660) || (c >= 32701 && c <= 32760) || (c >= 26901 && c <= 26923) || (c >= 25828 && c <= 25838) || (c >= 3395 && c <= 3396) || [27700, 2193, 3577, 3035, 5070, 3005, 2056, 28992, 31370].includes(c)) return ['metre', 1];
  return null;
}
function crsOf(code, def, geoFlag, name) {
  const c = code === null || code === undefined || code === '' || !isFinite(Number(code)) ? null : Number(code);
  const label = c !== null && c > 0 ? `EPSG:${c}` : name ? String(name) : 'no stated coordinate system';
  if (c !== null && MERC.has(c)) return { mode: 'merc', code: c, label };
  if (geoFlag === true || (c !== null && GEOG.has(c)) || /^\s*GEOG(?:CS|CRS)\s*\[/i.test(def || '')) return { mode: 'geo', code: c, label };
  const u = unitOfWkt(def) || unitOfCode(c);
  return { mode: 'native', code: c, label, unit: u ? u[0] : null, unitM: u ? u[1] : null };
}

        let crs;
        if (org === 'EPSG' && L.srsCode !== null) crs = crsOf(L.srsCode, L.srsDefinition);
        else if (Number(L.srsId) === 0) crs = { mode: 'geo', code: null, label: 'an undefined geographic coordinate system' };
        else if (Number(L.srsId) === -1) crs = { mode: 'native', code: -1, label: 'an undefined Cartesian coordinate system', unit: null, unitM: null };
        else crs = crsOf(null, L.srsDefinition, undefined, L.srsName || (L.srsId !== undefined ? `coordinate system ${L.srsId}` : ''));
        return { label: L.table, crs, features: L.features.map((f) => ({ props: f.props, geom: f.geom, error: f.error })) };
      });
    }
    if (fmt === 'geojson') { const g = geojsonOf(e); return [{ label: '', crs: g.crs, features: g.features.map((f) => ({ props: f.props, id: f.id, geom: f.geom })) }]; }
    if (fmt === 'tiff') {
      const t = await tiffOf(e);
      if (!t.geo || !t.geo.bounds) throw new Error('The picture holds no position on the Earth, so it can’t be placed on a map.');
      return [{ label: '', crs: crsOf(t.geo.epsg, '', t.geo.geographic === true), raster: t }];
    }
    throw new Error('This file has no map view.');
  });
}

  /* Every feature within about 10 pixels of a screen point, nearest first, so features drawn on top of each other are all
     listed. A point or line counts as nearer than the inside of a shape. */
  function hit(px, py) {
    const found = [], bd = 10.5;
    const segD = (ax, ay, bx, by) => { const dx = bx - ax, dy = by - ay, l = dx * dx + dy * dy, t = l ? Math.max(0, Math.min(1, ((px - ax) * dx + (py - ay) * dy) / l)) : 0; return Math.hypot(px - ax - t * dx, py - ay - t * dy); };
    for (let li = layers.length - 1; li >= 0; li--) {
      const L = layers[li];
      if (!L.on || L.off) continue;
      for (let fi = L.feats.length - 1; fi >= 0; fi--) {
        const f = L.feats[fi];
        let d = Infinity;
        for (const p of f.parts) {
          if (p.t === 'pt') { d = Math.min(d, Math.hypot(sx(p.r[0][0]) - px, sy(p.r[0][1]) - py)); continue; }
          let inside = false;
          for (const a of p.r) {
            const n = a.length / 2;
            for (let i = 0; i < n - 1 + (p.t === 'pg' ? 1 : 0); i++) {
              const j = (i + 1) % n, ax = sx(a[2 * i]), ay = sy(a[2 * i + 1]), bx = sx(a[2 * j]), by = sy(a[2 * j + 1]);
              d = Math.min(d, segD(ax, ay, bx, by));
              if (p.t === 'pg' && ((ay > py) !== (by > py)) && px < (bx - ax) * (py - ay) / (by - ay) + ax) inside = !inside;
            }
            if (n === 1) d = Math.min(d, Math.hypot(sx(a[0]) - px, sy(a[1]) - py));
          }
          if (inside) d = Math.min(d, 10);
        }
        if (d < bd) found.push({ L, f, d });
      }
    }
    return found.sort((a, b) => a.d - b.d);
  }

/* What each error code means, as the HTML standard defines it; nothing more is guessed. */
const MEDIA_ERRORS = {
  1: ['MEDIA_ERR_ABORTED', 'that reading the file was stopped before it finished'],
  2: ['MEDIA_ERR_NETWORK', 'an error while reading the file'],
  3: ['MEDIA_ERR_DECODE', 'an error while decoding it'],
  4: ['MEDIA_ERR_SRC_NOT_SUPPORTED', 'that the file, or its format, isn’t one it can play'],
};
function mediaWhy(err) {
  if (!err) return 'This browser can’t play the file, and it gives no reason.';
  const [name, says] = MEDIA_ERRORS[err.code] || ['', ''];
  const msg = err.message ? String(err.message).trim() : '';
  return `This browser can’t play the file. It reports ${says || 'an error'} (error code ${String(err.code)}${name ? ', ' + name : ''})${msg ? `, with the message “${msg}”` : ''}.`;
}
function lengthText(d) { // '2 min 05 s (125.373 seconds)', from the length the browser reports
  if (d === Infinity) return 'This browser reports no fixed length for it.';
  if (!isFinite(d)) return 'This browser reports no length for it.';
  const exact = d.toLocaleString('en-US', { maximumFractionDigits: 3 }), secs = `${exact} second${exact === '1' ? '' : 's'}`;
  if (d < 60) return `Length: ${secs}.`;
  const t = Math.round(d), h = Math.floor(t / 3600), m = Math.floor((t % 3600) / 60), s = t % 60, two = (n) => String(n).padStart(2, '0');
  return `Length: ${h ? `${fmtInt(h)} h ${two(m)} min` : `${m} min`} ${two(s)} s (${secs}).`;
}
function renderMedia(e, panel, tok, video) {
  const u8 = bytesOf(e);
  if (!u8.length) { panel.innerHTML = failNote(e, 'The file is empty, so there is nothing to play.'); return; }
  // one copy of the bytes for the player, made the first time and kept for the file, so going back
  // to this view doesn't need a second copy of a big file (the name alone gives its media type)
  const s = st(e);
  if (!s.mediaUrl) s.mediaUrl = URL.createObjectURL(new Blob([u8], { type: AUDIO_TYPES[extOf(e.name)] || VIDEO_TYPES[extOf(e.name)] || '' }));
  const url = s.mediaUrl;
  const tag = video ? 'video' : 'audio';
  // no autoplay, only the length is read until the person presses play, and it stays on this device
  panel.innerHTML = `
    <${tag} id="av-el" controls preload="metadata"${video ? ' playsinline' : ''} disableremoteplayback aria-label="${esc(`The ${video ? 'video' : 'sound'} in ${e.name}`)}" aria-describedby="av-meta">
    
    
    Reading the length…
    `;
  const el = $('av-el'), meta = $('av-meta');
  let done = false;
  const stop = () => { // pause and let go of the file; its address is freed by forget
    if (done) return;
    done = true;
    try { el.pause(); el.removeAttribute('src'); el.load(); } catch (err) { /* already stopped */ }
  };
  cleanup = stop;
  const describe = () => {
    if (!live(tok) || done) return;
    let t = lengthText(el.duration);
    if (video) t += el.videoWidth && el.videoHeight ? ` Picture: ${fmtInt(el.videoWidth)} × ${fmtInt(el.videoHeight)} pixels.` : ' This browser reports no picture in it.';
    if (meta.textContent !== t) meta.textContent = t;
  };
  el.addEventListener('loadedmetadata', describe);
  el.addEventListener('durationchange', () => { if (el.readyState >= 1) describe(); });
  if (video) el.addEventListener('resize', () => { if (el.readyState >= 1) describe(); });
  el.addEventListener('error', () => {
    if (!live(tok) || done) return;
    const why = mediaWhy(el.error);
    stop();
    panel.innerHTML = failNote(e, why);
  });
  el.src = url;
}

function fmtSize(n) {
  if (n < 1024) return plural(n, 'byte');
  const u = ['KB', 'MB', 'GB', 'TB'];
  let v = n / 1024, i = 0;
  while (v >= 1024 && i < u.length - 1) { v /= 1024; i++; }
  return `${v.toLocaleString('en-US', { maximumFractionDigits: v < 10 ? 1 : 0 })} ${u[i]}`;
}
